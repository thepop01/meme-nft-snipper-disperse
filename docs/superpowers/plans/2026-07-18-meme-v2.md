// Custom-list rule evaluation. All present rules AND together; blank rules
// are ignored. A rule whose required token data is missing FAILS (no false
// positives from incomplete data).

export const RULE_FIELDS = [
  { key: 'minLiquidityUsd', label: 'Min liquidity ($)', type: 'number' },
  { key: 'maxLiquidityUsd', label: 'Max liquidity ($)', type: 'number' },
  { key: 'minSafetyScore', label: 'Min safety score', type: 'number' },
  { key: 'minTractionScore', label: 'Min traction score', type: 'number' },
  { key: 'maxTop10HolderPct', label: 'Max top-10 holders (%)', type: 'number' },
  { key: 'maxAgeMin', label: 'Max age (minutes)', type: 'number' },
  { key: 'minVolume24hUsd', label: 'Min 24h volume ($)', type: 'number' },
  { key: 'source', label: 'Source', type: 'select', options: ['pumpfun', 'raydium'] },
  { key: 'keywordsInclude', label: 'Name/symbol contains any of', type: 'keywords' },
  { key: 'keywordsExclude', label: 'Name/symbol contains none of', type: 'keywords' },
  { key: 'minHolderCount', label: 'Min holder count', type: 'number' },
  { key: 'requireSocials', label: 'Require socials', type: 'boolean' },
];

const num = (x) => (typeof x === 'number' && isFinite(x) ? x : null);

export function matchesRules(token, rules = {}) {
  const reasons = [];
  const checks = {
    minLiquidityUsd: v => num(token.liquidityUsd) != null && token.liquidityUsd >= v,
    maxLiquidityUsd: v => num(token.liquidityUsd) != null && token.liquidityUsd <= v,
    minSafetyScore: v => num(token.safety?.score) != null && token.safety.score >= v,
    minTractionScore: v => num(token.traction?.tractionScore) != null && token.traction.tractionScore >= v,
    maxTop10HolderPct: v => num(token.top10HolderPct) != null && token.top10HolderPct <= v,
    maxAgeMin: v => token.createdAt != null && (Date.now() - token.createdAt) / 60_000 <= v,
    minVolume24hUsd: v => num(token.volume24hUsd) != null && token.volume24hUsd >= v,
    source: v => token.source === v,
    keywordsInclude: (v) => {
      const hay = `${token.symbol || ''} ${token.name || ''}`.toLowerCase();
      return (v || []).some(k => hay.includes(String(k).toLowerCase()));
    },
    keywordsExclude: (v) => {
      const hay = `${token.symbol || ''} ${token.name || ''}`.toLowerCase();
      return !(v || []).some(k => hay.includes(String(k).toLowerCase()));
    },
    minHolderCount: v => num(token.holderCount) != null && token.holderCount >= v,
    requireSocials: v => !v || Boolean(
      token.socials?.website || token.socials?.twitter || token.socials?.telegram),
  };

  for (const [key, value] of Object.entries(rules)) {
    if (value === undefined || value === null || value === '' ||
        (Array.isArray(value) && value.length === 0)) continue;
    const check = checks[key];
    if (!check) continue;
    if (!check(value)) return { match: false, reasons };
    reasons.push(key);
  }
  return { match: true, reasons };
}
```

// User-defined rule-based token lists. Evaluated on every analysis pass and
// refresh tick; matches are stamped with entry price and never silently
// dropped (exited flag instead).
import { load, save } from '../store.js';
import { emit } from '../bus.js';
import { pushAlert } from '../alerts.js';
import { matchesRules } from './listRules.js';

const STORE = 'custom-lists';
const COLORS = ['violet', 'green', 'blue', 'yellow', 'red'];

export function getLists() { return load(STORE, []); }
function persist(lists) { save(STORE, lists); }

export function createList({ name, rules = {}, color }) {
  const lists = getLists();
  const list = {
    id: `list_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
    name, color: color || COLORS[lists.length % COLORS.length],
    enabled: true, rules, matched: [], createdAt: Date.now(),
  };
  lists.push(list);
  persist(lists);
  return list;
}

export function updateList(id, patch) {
  const lists = getLists();
  const list = lists.find(l => l.id === id);
  if (!list) throw new Error('List not found');
  const { id: _ignore, matched: _keep, ...safe } = patch;
  Object.assign(list, safe);
  persist(lists);
  return list;
}

export function deleteList(id) {
  persist(getLists().filter(l => l.id !== id));
}

export function removeMatch(listId, mint) {
  const lists = getLists();
  const list = lists.find(l => l.id === listId);
  if (!list) throw new Error('List not found');
  list.matched = list.matched.filter(m => m.mint !== mint);
  persist(lists);
  return list;
}

// Custom-list evaluator: batch mode. Evaluates multiple tokens against all
// lists, loading and persisting the store only once.
export function evaluateTokenBatch(tokens) {
  const lists = getLists();
  let changed = false;
  for (const token of tokens) {
    for (const list of lists) {
      if (!list.enabled) continue;
      const { match, reasons } = matchesRules(token, list.rules);
      const entry = list.matched.find(m => m.mint === token.mint);
      if (match && !entry) {
        list.matched.push({
          mint: token.mint, listedAt: Date.now(),
          listedPriceUsd: token.priceUsd ?? null, exited: false, reasons,
        });
        changed = true;
        pushAlert({
          type: 'meme', severity: 'info',
          title: `${token.symbol || token.mint.slice(0, 6)} matched list "${list.name}"`,
          body: reasons.join(', '),
        });
        emit('token:listed', { listId: list.id, listName: list.name, mint: token.mint });
      } else if (entry && entry.exited !== !match) {
        entry.exited = !match;
        changed = true;
      }
    }
  }
  if (changed) persist(lists);
}

// Run one token through every enabled list. Called from registry passes and
// the refresh loop.
export function evaluateToken(token) {
  evaluateTokenBatch([token]);
}

// All mints matched by enabled lists (for the refresh loop).
export function trackedMints() {
  const out = new Set();
  for (const list of getLists()) {
    if (!list.enabled) continue;
    for (const m of list.matched) out.add(m.mint);
  }
  return [...out];
}
```

// Applies live market data to a token and emits token:update.
// patch: { priceUsd, liquidityUsd, marketCapUsd, volume24hUsd }
export function applyMarketPatch(mint, patch) {
  const token = tokens.get(mint);
  if (!token) return null;
  Object.assign(token, patch);
  token.history = token.history || [];
  token.history.push({ ts: Date.now(), priceUsd: patch.priceUsd ?? token.priceUsd,
    liquidityUsd: patch.liquidityUsd ?? token.liquidityUsd,
    marketCapUsd: patch.marketCapUsd ?? token.marketCapUsd });
  if (patch.liquidityUsd != null) {
    token.peakLiquidityUsd = Math.max(token.peakLiquidityUsd ?? 0, patch.liquidityUsd);
  }
  emit('token:update', { token });
  return token;
}

// Marks a curated token as rugged (liquidity collapse). Keeps it visible.
export function flagRugged(mint, reason) {
  const token = tokens.get(mint);
  if (!token || token.rugged) return null;
  token.rugged = true;
  token.ruggedReason = reason;
  emit('token:update', { token });
  return token;
}

```js
// Live-price refresh for curated tokens, custom-list matches, and open
// positions. One DexScreener batch (chunked at 30 mints, 500ms between
// chunks) per 45s tick; overlapping ticks are skipped.
import { log } from '../bus.js';
import { pushAlert } from '../alerts.js';
import {
  getCuratedTokens, getTokenByMint, applyMarketPatch, flagRugged,
} from './registry.js';
import { trackedMints, evaluateTokenBatch } from '../analysis/customLists.js';
import { getPositions } from '../engine/positions.js';

const TICK_MS = 45_000;
const CHUNK = 30;
const CHUNK_GAP_MS = 500;
const RUG_DROP_PCT = 70;
let running = false;

export function collectRefreshMints() {
  const out = new Set();
  for (const t of getCuratedTokens()) out.add(t.mint);
  for (const m of trackedMints()) out.add(m);
  for (const p of getPositions()) if (p.mint) out.add(p.mint);
  return [...out];
}

// Default price fetcher: DexScreener multi-token endpoint.
export async function fetchPricesFromDexScreener(mints) {
  const res = await fetch(`https://api.dexscreener.com/latest/dex/tokens/${mints.join(',')}`);
  if (!res.ok) throw new Error(`DexScreener ${res.status}`);
  const data = await res.json();
  const mintSet = new Set(mints); // O(1) lookups instead of O(n) includes()
  const best = new Map(); // mint -> highest-liquidity pair
  for (const pair of data.pairs || []) {
    const mint = pair.baseToken?.address;
    if (!mint || !mintSet.has(mint)) continue;
    const liq = pair.liquidity?.usd ?? 0;
    if (!best.has(mint) || liq > (best.get(mint).liquidity?.usd ?? 0)) best.set(mint, pair);
  }
  return [...best.entries()].map(([mint, pair]) => ({
    mint,
    priceUsd: pair.priceUsd != null ? Number(pair.priceUsd) : null,
    liquidityUsd: pair.liquidity?.usd ?? null,
    marketCapUsd: pair.marketCap ?? pair.fdv ?? null,
    volume24hUsd: pair.volume?.h24 ?? null,
  }));
}

export async function runRefreshTick({ fetchPrices = fetchPricesFromDexScreener } = {}) {
  if (running) return;
  running = true;
  try {
    const mints = collectRefreshMints();
    const patchedTokens = [];
    for (let i = 0; i < mints.length; i += CHUNK) {
      const chunk = mints.slice(i, i + CHUNK);
      let prices;
      try {
        prices = await fetchPrices(chunk);
      } catch (err) {
        log('warn', `Price refresh chunk failed: ${err.message}`);
        continue;
      }
      for (const p of prices) {
        const before = getTokenByMint(p.mint);
        if (!before) continue;
        const peak = before.peakLiquidityUsd ?? 0;
        const token = applyMarketPatch(p.mint, {
          priceUsd: p.priceUsd, liquidityUsd: p.liquidityUsd,
          marketCapUsd: p.marketCapUsd, volume24hUsd: p.volume24hUsd,
        });
        if (!token) continue;
        patchedTokens.push(token);
        // Rug detection against the pre-patch peak.
        if (!token.rugged && peak > 0 && p.liquidityUsd != null &&
            p.liquidityUsd < peak * (1 - RUG_DROP_PCT / 100)) {
          flagRugged(p.mint, `liquidity -${RUG_DROP_PCT}% from peak`);
          pushAlert({
            type: 'meme', severity: 'critical',
            title: `${token.symbol || p.mint.slice(0, 6)} flagged as rugged`,
            body: `Liquidity collapsed from $${Math.round(peak)} to $${Math.round(p.liquidityUsd)}`,
          });
        }
      }
      if (i + CHUNK < mints.length) await new Promise(r => setTimeout(r, CHUNK_GAP_MS));
    }
    // Batch-evaluate all patched tokens against custom lists (single load/persist).
    if (patchedTokens.length > 0) evaluateTokenBatch(patchedTokens);
  } finally {
    running = false;
  }
}

export function startRefreshLoop() {
  setInterval(() => runRefreshTick().catch(err =>
    log('error', `Refresh tick failed: ${err.message}`)), TICK_MS);
}

// Since-curated / since-listed performance math.

export function pctChange(entryPrice, currentPrice) {
  if (!entryPrice || currentPrice == null) return null;
  return ((currentPrice - entryPrice) / entryPrice) * 100;
}

// entries: [{ entryPrice, currentPrice, ts }]
export function computeStats(entries, { windowMs, now = Date.now() }) {
  const pcts = entries
    .filter(e => now - e.ts <= windowMs)
    .map(e => pctChange(e.entryPrice, e.currentPrice))
    .filter(p => p != null)
    .sort((a, b) => a - b);
  if (pcts.length === 0) {
    return { count: 0, medianPct: null, winRatePct: null, bestPct: null, worstPct: null };
  }
  const mid = Math.floor(pcts.length / 2);
  const medianPct = pcts.length % 2 ? pcts[mid] : (pcts[mid - 1] + pcts[mid]) / 2;
  return {
    count: pcts.length,
    medianPct,
    winRatePct: (pcts.filter(p => p > 0).length / pcts.length) * 100,
    bestPct: pcts[pcts.length - 1],
    worstPct: pcts[0],
  };
}
```

export function createListRouter() {
  const router = Router();
  const wrap = (fn) => (req, res) =>
    Promise.resolve(fn(req, res)).catch(err => res.status(400).json({ error: err.message }));

  router.get('/', (req, res) => res.json({ lists: getLists(), ruleFields: RULE_FIELDS }));
  router.post('/', wrap((req, res) => res.json({ list: createList(req.body || {}) })));

  router.delete('/:id', wrap((req, res) => { deleteList(req.params.id); res.json({ ok: true }); }));
  router.delete('/:id/matches/:mint', wrap((req, res) =>
    res.json({ list: removeMatch(req.params.id, req.params.mint) })));

  // Tokens of one list, enriched with live data + % since listed.
  router.get('/:id/tokens', wrap((req, res) => {
    const list = getLists().find(l => l.id === req.params.id);
    if (!list) throw new Error('List not found');
    const tokens = list.matched.map(m => {
      const t = getTokenByMint(m.mint);
      return t ? {
        ...t,
        listedAt: m.listedAt, listedPriceUsd: m.listedPriceUsd, exited: m.exited,
        matchReasons: m.reasons || [],
        sinceListedPct: pctChange(m.listedPriceUsd, t.priceUsd),
      } : null;
    }).filter(Boolean);
    res.json({ tokens });
  }));

  router.get('/:id/stats', wrap((req, res) => {
    const list = getLists().find(l => l.id === req.params.id);
    if (!list) throw new Error('List not found');
    const entries = list.matched.map(m => {
      const t = getTokenByMint(m.mint);
      return { entryPrice: m.listedPriceUsd, currentPrice: t?.priceUsd ?? null, ts: m.listedAt };
    });
    res.json({
      stats24h: computeStats(entries, { windowMs: WINDOWS['24h'] }),
      stats7d: computeStats(entries, { windowMs: WINDOWS['7d'] }),
    });
  }));

  return router;
}
